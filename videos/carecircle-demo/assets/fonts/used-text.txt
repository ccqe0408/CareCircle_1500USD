<script>
  (function () {
    "use strict";

    var vars =
      window.__hyperframes && window.__hyperframes.getVariables
        ? window.__hyperframes.getVariables()
        : {};

    // Unrecognised overrides return to their declared defaults.
    var axes = {
      up: { x: 0, y: 1 },
      down: { x: 0, y: -1 },
      left: { x: -1, y: 0 },
      right: { x: 1, y: 0 },
    };
    var reaches = { close: 0.5, standard: 1, far: 1.85 };
    var tones = {
      ink: "var(--fg, #18181b)",
      paper: "#fafafa",
      accent: "var(--brand, #34d399)",
    };

    function pick(table, value, fallback) {
      return Object.prototype.hasOwnProperty.call(table, value) ? value : fallback;
    }

    var axis = axes[pick(axes, vars.direction, "up")];
    var reach = reaches[pick(reaches, vars.distance, "standard")];
    var tone = tones[pick(tones, vars.tone, "ink")];
    var words =
      typeof vars.words === "string" && vars.words.trim() ? vars.words.trim().split(/\s+/) : null;

    var roots = document.querySelectorAll(".hf-catalog-staggered-fade-up");
    for (var i = 0; i < roots.length; i += 1) {
      roots[i].style.setProperty("--hf-word-axis-x", String(axis.x * reach));
      roots[i].style.setProperty("--hf-word-axis-y", String(axis.y * reach));
      roots[i].style.setProperty("--hf-fade-word-color", tone);
      if (words) {
        roots[i].textContent = "";
        for (var j = 0; j < words.length; j += 1) {
          var span = document.createElement("span");
          span.textContent = words[j];
          roots[i].appendChild(span);
        }
      }
    }
  })();
</script>
